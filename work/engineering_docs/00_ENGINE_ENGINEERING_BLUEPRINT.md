  const latinStr = buf.toString('latin1');

  // 1. Carve embedded HTML documents (<!DOCTYPE html> or <html ... </html>)

  let htmlIdx = 0;

      htmlIdx++;

      fs.writeFileSync(outPath, rawHtml, 'utf8');
